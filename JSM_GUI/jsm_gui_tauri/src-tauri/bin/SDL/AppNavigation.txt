# JSM Studio: Studio has the controller
#
# Auto-loaded while JSM Studio is the foreground window (an AutoLoad rule named
# after the app's own executable points here). Studio reads the pad itself for
# navigation, from the same telemetry stream that draws the live preview, so
# this profile maps nothing to keys: the configuration you were playing with is
# paused, and every standard button is free for Studio.
#
#   D-pad / left stick   move            A / B     select / back
#   X / Y                toggle / details LB / RB  previous / next section
#   LT / RT              previous / next page
#   View / Menu          title bar / page actions
#   Right stick          scroll          Hold B    escape from capture
#
# Only the buttons navigate: the trackpads and the gyro do nothing here, so the
# pointer never moves on its own while Studio has the controller. Global chords
# are handled by Studio from telemetry, so they still work here. Starts with
# RESET_MAPPINGS so OnReset.txt still applies. Test mode (the Test button in
# the title bar) loads the configuration instead, until View + Menu is held,
# Esc is pressed or Return to Studio is clicked; alt-tabbing to a game
# restores that game's own configuration as before.

RESET_MAPPINGS

# Studio's required settings, as in every profile it writes. RESET_MAPPINGS
# turns telemetry off, and telemetry is how Studio sees the controller at all:
# without these lines it shows "No controller", navigation stops and global
# chords stop with it.
AUTOCONNECT = ON
TELEMETRY_ENABLED = ON
TELEMETRY_PORT = 8974
StudioDefaults.txt

LEFT_STICK_MODE = NO_MOUSE
RIGHT_STICK_MODE = NO_MOUSE

# Pads: a grid with nothing bound, and touch sticks that move nothing (the
# defaults, said out loud so a changed default cannot bring the mouse back).
LEFT_TOUCHPAD_MODE = GRID_AND_STICK
RIGHT_TOUCHPAD_MODE = GRID_AND_STICK
LEFT_TOUCH_STICK_MODE = NO_MOUSE
RIGHT_TOUCH_STICK_MODE = NO_MOUSE

# Gyro always off.
GYRO_ON = NONE
