# JSM Studio global chord layer
#
# Loaded by OnReset.txt, so it applies on top of EVERY configuration the moment
# it loads (including auto-switches). Hold the Steam button (HOME) or the Quick
# Access "..." button (MISC1) and press another input to use these, the same
# way Steam Input's Guide-button chords work. A configuration can still
# override any line here by setting the same chord itself.
#
# Edit from the "Global chords" page in JSM Studio, or by hand. Do NOT put
# RESET_MAPPINGS in this file -- it would loop.

# The modifiers do nothing on their own.
HOME = NONE
MISC1 = NONE

# Power the controller off: Steam+Y or ...+Y (the same two chords as Steam).
HOME,Y = TURN_OFF_CONTROLLER
MISC1,Y = TURN_OFF_CONTROLLER

# While Steam is held the right pad and stick drive the desktop cursor and the
# pad clicks are the mouse buttons.
HOME,RIGHT_TOUCHPAD_MODE = MOUSE
HOME,RIGHT_STICK_MODE = AIM
HOME,MISC2 = LMOUSE
HOME,MISC3 = RMOUSE

# Left stick as a volume rocker, click to mute.
HOME,LEFT_STICK_MODE = NO_MOUSE
HOME,LUP = VOLUME_UP
HOME,LDOWN = VOLUME_DOWN
HOME,L3 = MUTE

# Window switching: L4 = Alt+Tab, R4 = Windows key.
HOME,LSL = LALT\ !TAB\
HOME,RSR = LWINDOWS

# Confirm / back while a chord menu is up.
HOME,X = ENTER
HOME,B = ESC
