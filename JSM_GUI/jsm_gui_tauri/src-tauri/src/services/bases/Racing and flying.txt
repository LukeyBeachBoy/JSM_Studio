# Racing and flying. Games see a virtual Xbox controller, so the triggers
# keep their full analog range for throttle and brake. It ships with JSM
# Evolved, so it can't be changed in place: duplicate it to make your own.
VIRTUAL_CONTROLLER = XBOX

# Steering on the left stick, the camera on the right
LEFT_STICK_MODE = LEFT_STICK
RIGHT_STICK_MODE = RIGHT_STICK

# Analog throttle and brake
ZR_MODE = X_RT
ZL_MODE = X_LT

# Every button as it is on an Xbox pad
S = X_A
E = X_B
W = X_X
N = X_Y
L = X_LB
R = X_RB
+ = X_START
- = X_BACK
L3 = X_LS
R3 = X_RS
UP = X_UP
DOWN = X_DOWN
LEFT = X_LEFT
RIGHT = X_RIGHT

# @label ZR = Throttle
# @label ZL = Brake
# @label S = Handbrake
# @label E = Look back
# @label L = Shift down
# @label R = Shift up
